## 5. PasswordInput — input com toggle de visibilidade

```tsx
// src/components/auth/PasswordInput.tsx
'use client';

import { useState } from "react";
import { Eye, EyeOff } from "lucide-react";
import { forwardRef } from "react";

type PasswordInputProps = React.InputHTMLAttributes<HTMLInputElement> & {
  label: string;
  error?: string;
  rightLabel?: React.ReactNode; // slot para "Esqueceu a senha?"
};

export const PasswordInput = forwardRef<HTMLInputElement, PasswordInputProps>(
  ({ label, error, rightLabel, id, ...props }, ref) => {
    const [show, setShow] = useState(false);
    const inputId = id ?? "password";
    const errorId = `${inputId}-error`;

    return (
      <div className="space-y-1">
        <div className="flex items-center justify-between">
          <label htmlFor={inputId} className="text-sm font-medium text-text-secondary">
            {label}
          </label>
          {rightLabel}
        </div>
        <div className="relative">
          <input
            ref={ref}
            id={inputId}
            type={show ? "text" : "password"}
            aria-invalid={!!error}
            aria-describedby={error ? errorId : undefined}
            className="w-full rounded-lg border border-border-muted bg-bg-base px-3 py-2 pr-10 text-text-primary focus:border-brand-primary focus:outline-none focus:ring-1 focus:ring-brand-primary aria-invalid:border-status-error"
            {...props}
          />
          <button
            type="button"
            onClick={() => setShow((v) => !v)}
            aria-label={show ? "Ocultar senha" : "Mostrar senha"}
            className="absolute right-3 top-1/2 -translate-y-1/2 text-text-muted hover:text-text-secondary"
          >
            {show ? (
              <EyeOff className="h-4 w-4" aria-hidden="true" />
            ) : (
              <Eye className="h-4 w-4" aria-hidden="true" />
            )}
          </button>
        </div>
        {error && (
          <p id={errorId} className="text-xs text-status-error-text" role="alert">
            {error}
          </p>
        )}
      </div>
    );
  }
);

PasswordInput.displayName = "PasswordInput";
```

## 6. LoginForm

```tsx
// src/components/auth/LoginForm.tsx
'use client';

import { useRef, useState, useEffect } from "react";
import { useForm } from "react-hook-form";
import { zodResolver } from "@hookform/resolvers/zod";
import { useRouter } from "next/navigation";
import Link from "next/link";
import { useTranslations } from "next-intl";
import { Loader2 } from "lucide-react";

import { createClient } from "@/lib/supabase/client";
import { loginSchema, type LoginFormValues } from "@/lib/auth/schemas";
import { mapSupabaseAuthError } from "@/lib/auth/errors";
import { AuthCard } from "./AuthCard";
import { AuthErrorBanner } from "./AuthErrorBanner";
import { PasswordInput } from "./PasswordInput";

type LoginFormProps = {
  showRegisteredToast?: boolean;
};

export function LoginForm({ showRegisteredToast }: LoginFormProps) {
  const t = useTranslations("auth");
  const router = useRouter();
  const errorBannerRef = useRef<HTMLDivElement>(null);

  const [isLoading, setIsLoading] = useState(false);
  const [submitError, setSubmitError] = useState<string | null>(null);

  const {
    register,
    handleSubmit,
    watch,
    formState: { errors },
  } = useForm<LoginFormValues>({
    resolver: zodResolver(loginSchema),
  });

  // Limpa erro ao editar campos
  const watched = watch();
  useEffect(() => {
    if (submitError) setSubmitError(null);
  }, [watched.email, watched.password]); // eslint-disable-line react-hooks/exhaustive-deps

  // Foco no banner quando erro aparece
  useEffect(() => {
    if (submitError) errorBannerRef.current?.focus();
  }, [submitError]);

  // Toast pós-cadastro
  useEffect(() => {
    if (showRegisteredToast) {
      // integrar com sistema de toast do projeto (shadcn/ui Toast ou similar)
      // toast.success(t("toast.account_created"));
    }
  }, [showRegisteredToast, t]);

  const onSubmit = async (values: LoginFormValues) => {
    setIsLoading(true);
    setSubmitError(null);

    const supabase = createClient();
    const { error } = await supabase.auth.signInWithPassword({
      email: values.email,
      password: values.password,
    });

    if (error) {
      setSubmitError(t(mapSupabaseAuthError(error)));
      setIsLoading(false);
      return;
    }

    router.push("/upload");
    router.refresh();
  };

  return (
    <AuthCard subtitle={t("login.subtitle")}>
      <h1 className="mb-1 text-lg font-semibold text-text-primary">
        {t("login.heading")}
      </h1>
      <p className="mb-5 text-sm text-text-secondary">{t("login.subheading")}</p>

      {submitError && (
        <AuthErrorBanner ref={errorBannerRef} message={submitError} />
      )}

      <form onSubmit={handleSubmit(onSubmit)} noValidate className="space-y-4">
        {/* Email */}
        <div className="space-y-1">
          <label htmlFor="email" className="text-sm font-medium text-text-secondary">
            {t("fields.email")}
          </label>
          <input
            id="email"
            type="email"
            autoComplete="email"
            placeholder="nome@empresa.com.br"
            aria-invalid={!!errors.email}
            aria-describedby={errors.email ? "email-error" : undefined}
            className="w-full rounded-lg border border-border-muted bg-bg-base px-3 py-2 text-text-primary focus:border-brand-primary focus:outline-none focus:ring-1 focus:ring-brand-primary aria-invalid:border-status-error"
            {...register("email")}
          />
          {errors.email && (
            <p id="email-error" role="alert" className="text-xs text-status-error-text">
              {t(errors.email.message!)}
            </p>
          )}
        </div>

        {/* Senha */}
        <PasswordInput
          id="password"
          label={t("fields.password")}
          autoComplete="current-password"
          error={errors.password ? t(errors.password.message!) : undefined}
          rightLabel={
            <button
              type="button"
              disabled
              title={t("forgot_password.tooltip")}
              className="cursor-not-allowed text-xs text-brand-primary opacity-50"
              aria-disabled="true"
            >
              {t("forgot_password.label")}
            </button>
          }
          {...register("password")}
        />

        {/* Submit */}
        <button
          type="submit"
          disabled={isLoading}
          aria-busy={isLoading}
          className="flex w-full items-center justify-center gap-2 rounded-lg bg-brand-primary py-2.5 text-sm font-medium text-white transition-opacity hover:opacity-90 disabled:cursor-not-allowed disabled:opacity-60"
        >
          {isLoading && <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" />}
          {t("login.cta")}
        </button>
      </form>

      {/* Link para cadastro */}
      <p className="mt-5 text-center text-sm text-text-secondary">
        {t("login.no_account")}{" "}
        <Link href="/cadastro" className="font-medium text-brand-primary hover:underline">
          {t("login.signup_link")}
        </Link>
      </p>
    </AuthCard>
  );
}
```

## 7. SignupForm

```tsx
// src/components/auth/SignupForm.tsx
'use client';

import { useRef, useState, useEffect } from "react";
import { useForm } from "react-hook-form";
import { zodResolver } from "@hookform/resolvers/zod";
import { useRouter } from "next/navigation";
import Link from "next/link";
import { useTranslations } from "next-intl";
import { Loader2, ShieldCheck } from "lucide-react";

import { createClient } from "@/lib/supabase/client";
import { signupSchema, type SignupFormValues } from "@/lib/auth/schemas";
import { mapSupabaseAuthError } from "@/lib/auth/errors";
import { AuthCard } from "./AuthCard";
import { AuthErrorBanner } from "./AuthErrorBanner";
import { PasswordInput } from "./PasswordInput";

export function SignupForm() {
  const t = useTranslations("auth");
  const router = useRouter();
  const errorBannerRef = useRef<HTMLDivElement>(null);

  const [isLoading, setIsLoading] = useState(false);
  const [submitError, setSubmitError] = useState<string | null>(null);

  const {
    register,
    handleSubmit,
    watch,
    formState: { errors },
  } = useForm<SignupFormValues>({
    resolver: zodResolver(signupSchema),
  });

  const watched = watch();
  useEffect(() => {
    if (submitError) setSubmitError(null);
  }, [watched.email, watched.password, watched.confirmPassword]); // eslint-disable-line react-hooks/exhaustive-deps

  useEffect(() => {
    if (submitError) errorBannerRef.current?.focus();
  }, [submitError]);

  const onSubmit = async (values: SignupFormValues) => {
    setIsLoading(true);
    setSubmitError(null);

    const supabase = createClient();
    const { error } = await supabase.auth.signUp({
      email: values.email,
      password: values.password,
    });

    if (error) {
      setSubmitError(t(mapSupabaseAuthError(error)));
      setIsLoading(false);
      return;
    }

    router.push("/login?registered=true");
  };

  return (
    <AuthCard subtitle={t("signup.subtitle")}>
      <h1 className="mb-5 text-lg font-semibold text-text-primary">
        {t("signup.heading")}
      </h1>

      {submitError && (
        <AuthErrorBanner ref={errorBannerRef} message={submitError} />
      )}

      <form onSubmit={handleSubmit(onSubmit)} noValidate className="space-y-4">
        {/* Email */}
        <div className="space-y-1">
          <label htmlFor="email" className="text-xs font-semibold uppercase tracking-wide text-text-secondary">
            {t("fields.email")}
          </label>
          <input
            id="email"
            type="email"
            autoComplete="email"
            placeholder="nome@empresa.com.br"
            aria-invalid={!!errors.email}
            aria-describedby={errors.email ? "email-error" : undefined}
            className="w-full rounded-lg border border-border-muted bg-bg-base px-3 py-2 text-text-primary focus:border-brand-primary focus:outline-none focus:ring-1 focus:ring-brand-primary aria-invalid:border-status-error"
            {...register("email")}
          />
          {errors.email && (
            <p id="email-error" role="alert" className="text-xs text-status-error-text">
              {t(errors.email.message!)}
            </p>
          )}
        </div>

        {/* Senha */}
        <PasswordInput
          id="password"
          label={t("fields.password_signup")}
          autoComplete="new-password"
          error={errors.password ? t(errors.password.message!) : undefined}
          {...register("password")}
        />

        {/* Confirmar senha */}
        <PasswordInput
          id="confirmPassword"
          label={t("fields.confirm_password")}
          autoComplete="new-password"
          error={errors.confirmPassword ? t(errors.confirmPassword.message!) : undefined}
          {...register("confirmPassword")}
        />

        {/* Submit */}
        <button
          type="submit"
          disabled={isLoading}
          aria-busy={isLoading}
          className="flex w-full items-center justify-center gap-2 rounded-lg bg-brand-primary py-2.5 text-sm font-medium text-white transition-opacity hover:opacity-90 disabled:cursor-not-allowed disabled:opacity-60"
        >
          {isLoading && <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" />}
          {t("signup.cta")}
        </button>
      </form>

      {/* Link para login */}
      <p className="mt-4 text-center text-sm text-text-secondary">
        {t("signup.has_account")}{" "}
        <Link href="/login" className="font-medium text-brand-primary hover:underline">
          {t("signup.login_link")}
        </Link>
      </p>

      {/* LGPD */}
      <div className="mt-4 flex items-center justify-center gap-1.5 text-xs text-text-muted">
        <ShieldCheck className="h-3.5 w-3.5 shrink-0" aria-hidden="true" />
        <span>{t("signup.lgpd_note")}</span>
      </div>
    </AuthCard>
  );
}
```
