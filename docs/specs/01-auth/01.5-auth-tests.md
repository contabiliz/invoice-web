## 2. Testes funcionais — LoginForm

### T-FUNC-01 — Login com credenciais válidas redireciona para `/upload`

```ts
it("redirects to /upload on successful login", async () => {
  const mockPush = vi.fn();
  vi.mocked(useRouter).mockReturnValue({ push: mockPush, refresh: vi.fn() } as unknown as AppRouterInstance);

  vi.mocked(createClient).mockReturnValue({
    auth: { signInWithPassword: mockSignInSuccess() },
  } as unknown as SupabaseClient);

  render(<LoginForm />);

  await userEvent.type(screen.getByLabelText(/email/i), validCredentials.email);
  await userEvent.type(screen.getByLabelText(/senha/i), validCredentials.password);
  await userEvent.click(screen.getByRole("button", { name: /entrar/i }));

  await waitFor(() => {
    expect(mockPush).toHaveBeenCalledWith("/upload");
  });
});
```

### T-FUNC-02 — Credenciais inválidas exibem banner PT-BR

```ts
it("shows PT-BR error banner for invalid credentials", async () => {
  vi.mocked(createClient).mockReturnValue({
    auth: {

    },
  } as unknown as SupabaseClient);

  render(<LoginForm />);

  await waitFor(() => {
    const alert = screen.getByRole("alert");
    expect(alert).toHaveTextContent("Email ou senha incorretos");
    expect(alert).not.toHaveTextContent("Invalid login credentials"); // raw nunca aparece
  });
});
```

### T-FUNC-03 — Banner some ao editar campo após erro

```ts
it("clears error banner when user edits email after error", async () => {
  // setup: exibir erro
  vi.mocked(createClient).mockReturnValue({
    auth: { signInWithPassword: mockSignInError("Invalid login credentials") },
  } as unknown as SupabaseClient);

  render(<LoginForm />);
  await userEvent.type(screen.getByLabelText(/email/i), "errado@a.com");
  await userEvent.type(screen.getByLabelText(/senha/i), "errado");
  await userEvent.click(screen.getByRole("button", { name: /entrar/i }));
  await screen.findByRole("alert");

  // editar campo → banner deve sumir
  await userEvent.clear(screen.getByLabelText(/email/i));
  await userEvent.type(screen.getByLabelText(/email/i), "novo@a.com");

  expect(screen.queryByRole("alert")).not.toBeInTheDocument();
});
```

## 5. Testes do mapeamento de erros

```ts
// src/lib/auth/__tests__/errors.test.ts
import { mapSupabaseAuthError } from "@/lib/auth/errors";
import type { AuthError } from "@supabase/supabase-js";

const makeError = (message: string, status?: number): AuthError =>
  ({ message, status, name: "AuthApiError" }) as AuthError;

describe("mapSupabaseAuthError", () => {
  it("maps Invalid login credentials", () => {
    expect(mapSupabaseAuthError(makeError("Invalid login credentials")))
      .toBe("errors.invalid_credentials");
  });

  it("maps Email not confirmed", () => {
    expect(mapSupabaseAuthError(makeError("Email not confirmed")))
      .toBe("errors.email_not_confirmed");
  });

  it("maps User already registered", () => {
    expect(mapSupabaseAuthError(makeError("User already registered")))
      .toBe("errors.email_already_registered");
  });

  it("maps 429 rate limit", () => {
    expect(mapSupabaseAuthError(makeError("Too many requests", 429)))
      .toBe("errors.rate_limited");
  });

  it("maps 500 server error", () => {
    expect(mapSupabaseAuthError(makeError("Internal server error", 500)))
      .toBe("errors.server_error");
  });

  it("maps unknown errors to unexpected", () => {
    expect(mapSupabaseAuthError(makeError("Something weird happened", 400)))
      .toBe("errors.unexpected");
  });
});
```

## 6. Testes de middleware

```ts
// src/middleware.test.ts
import { middleware } from "@/middleware";
import { NextRequest } from "next/server";

// Mock do @supabase/ssr
vi.mock("@supabase/ssr", () => ({
  createServerClient: vi.fn(() => ({
    auth: {
      getUser: vi.fn(),
    },
  })),
}));

const makeRequest = (pathname: string) =>
  new NextRequest(new URL(pathname, "http://localhost:3000"));

it("redirects unauthenticated user from /upload to /login", async () => {
  vi.mocked(createServerClient).mockReturnValue({
    auth: { getUser: vi.fn().mockResolvedValue({ data: { user: null } }) },
  } as unknown as ReturnType<typeof createServerClient>);

  const req = makeRequest("/upload");
  const res = await middleware(req);

  expect(res.status).toBe(307);
  expect(res.headers.get("location")).toContain("/login");
  expect(res.headers.get("location")).toContain("callbackUrl=%2Fupload");
});

it("redirects authenticated user from /login to /upload", async () => {
  vi.mocked(createServerClient).mockReturnValue({
    auth: {
      getUser: vi.fn().mockResolvedValue({ data: { user: { id: "uuid" } } }),
    },
  } as unknown as ReturnType<typeof createServerClient>);

  const req = makeRequest("/login");
  const res = await middleware(req);

  expect(res.status).toBe(307);
  expect(res.headers.get("location")).toContain("/upload");
});

it("allows unauthenticated access to /cadastro", async () => {
  vi.mocked(createServerClient).mockReturnValue({
    auth: { getUser: vi.fn().mockResolvedValue({ data: { user: null } }) },
  } as unknown as ReturnType<typeof createServerClient>);

  const req = makeRequest("/cadastro");
  const res = await middleware(req);

  expect(res.status).toBe(200);
});
```
